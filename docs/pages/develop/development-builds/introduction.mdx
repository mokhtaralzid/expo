// مكونات مساعدة

  <View style={styles.featureItem}>
    <Ionicons name={icon} size={24} color="#1e6a8c" />
    <Text style={styles.featureText}>{title}</Text>
  </View>

const LessonCard = ({ title, duration, level }) => (
  <TouchableOpacity style={styles.lessonCard}>

      <Text style={styles.lessonDetail}>{level}</Text>
    </View>
  </TouchableOpacity>
);

// تكوين التنقل
const Stack = createStackNavigator();
const Tab = createBottomTabNavigator();

const HomeStack = () => (
  <Stack.Navigator>
    <Stack.Screen name="الرئيسية" component={HomeScreen} />
    <Stack.Screen name="الدروس" component={LessonsScreen} />
    <Stack.Screen name="المجتمع" component={CommunityScreen} />
  </Stack.Navigator>
);

export default function App() {
  return (
    <NavigationContainer>
      <Tab.Navigator
        screenOptions={({ route }) => ({
          tabBarIcon: ({ color, size }) => {
            let iconName;
            if (route.name === 'الرئيسية') iconName = 'home';
            else if (route.name === 'الدروس') iconName = 'book';
            else if (route.name === 'المجتمع') iconName = 'people';
            else if (route.name === 'الملف') iconName = 'person';
            return <Ionicons name={iconName} size={size} color={color} />;
          },
          tabBarActiveTintColor: '#1e6a8c',
          tabBarInactiveTintColor: 'gray',
          tabBarLabelStyle: { fontFamily: 'NotoSansArabic' },
          tabBarStyle: { paddingVertical: 5 }
        })}
      >
        <Tab.Screen name="الرئيسية" component={HomeStack} />
        <Tab.Screen name="الدروس" component={LessonsScreen} />
        <Tab.Screen name="المجتمع" component={CommunityScreen} />
        <Tab.Screen name="الملف" component={ProfileScreen} />
      </Tab.Navigator>
    </NavigationContainer>
  );
}
